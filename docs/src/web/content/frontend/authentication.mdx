In v3, frontend auth state belongs to your app. The browser signs in through your framework routes, then calls your own API routes with cookies or headers. Those server routes validate the user and call Agentuity services with server-only credentials.

```tsx
import { useEffect, useState } from 'react';

interface AppUser {
  readonly id: string;
  readonly email: string | null;
  readonly name: string | null;
}

interface SessionResponse {
  readonly authenticated: boolean;
  readonly user: AppUser | null;
}

function readAppUser(value: unknown): AppUser | null {
  if (typeof value !== 'object' || value === null) {
    return null;
  }

  if (!('id' in value) || typeof value.id !== 'string') {
    return null;
  }

  if (!('email' in value) || (typeof value.email !== 'string' && value.email !== null)) {
    return null;
  }

  if (!('name' in value) || (typeof value.name !== 'string' && value.name !== null)) {
    return null;
  }

  return {
    id: value.id,
    email: value.email,
    name: value.name,
  };
}

function readSessionResponse(value: unknown): SessionResponse {
  if (typeof value !== 'object' || value === null) {
    return { authenticated: false, user: null };
  }

  if (!('authenticated' in value) || typeof value.authenticated !== 'boolean') {
    return { authenticated: false, user: null };
  }

  if (!value.authenticated || !('user' in value)) {
    return { authenticated: false, user: null };
  }

  const user = readAppUser(value.user);

  if (!user) {
    return { authenticated: false, user: null };
  }

  return {
    authenticated: true,
    user,
  };
}

export function AccountMenu() {
  const [session, setSession] = useState<SessionResponse | null>(null);

  useEffect(() => {
    let active = true;

    async function loadSession(): Promise<void> {
      const response = await fetch('/auth/me', { credentials: 'include' });
      const nextSession = readSessionResponse(await response.json());

      if (active) {
        setSession(nextSession);
      }
    }

    void loadSession();

    return () => {
      active = false;
    };
  }, []);

  if (!session) {
    return <button disabled>Checking session</button>;
  }

  if (!session.authenticated) {
    return <a href="/auth/login">Sign in with Agentuity</a>;
  }

  return (
    <form method="post" action="/auth/logout">
      <span>{session.user?.email}</span>
      <button type="submit">Sign out</button>
    </form>
  );
}
```

The component above does not import an Agentuity auth provider. It reads your app's session endpoint and sends the user to your app's login route.

## Server Routes Own the Session

Keep the auth provider, session lookup, and Agentuity service calls on the server side.

```typescript
import { KeyValueClient } from '@agentuity/keyvalue';

interface AppUser {
  readonly id: string;
  readonly email: string;
}

interface AppSession {
  readonly user: AppUser;
}

interface DashboardState {
  readonly lastOpenedAt: string;
}

declare function getSession(request: Request): Promise<AppSession | null>;

const kv = new KeyValueClient();

export async function GET(request: Request): Promise<Response> {
  const session = await getSession(request);

  if (!session) {
    return Response.json({ authenticated: false }, { status: 401 });
  }

  const result = await kv.get<DashboardState>('dashboard-state', session.user.id);

  return Response.json({
    authenticated: true,
    user: session.user,
    state: result.exists ? result.data : null,
  });
}
```

Use your framework's normal route shape for this code. In Next.js this is a route handler. In TanStack Start, Remix, SvelteKit, Nuxt, Astro, or Hono, the same work happens in the server route for that framework.

## Add Sign in with Agentuity

When Agentuity is the identity provider, the frontend only needs links or forms to your auth routes:

```tsx
export function SignedOutActions() {
  return (
    <div>
      <a href="/auth/login">Sign in with Agentuity</a>
    </div>
  );
}
```

The `/auth/login`, `/auth/callback`, `/auth/me`, and `/auth/logout` routes live on the server. See [Sign in with Agentuity](/services/oidc-provider): it shows the OAuth redirect, callback state check, token exchange, and local app session.

## Protect Browser Calls

- Call your own API routes from the browser.
- Validate the session in each route that needs a signed-in user.
- Keep `AGENTUITY_SDK_KEY`, `OAUTH_CLIENT_SECRET`, and refresh tokens out of browser bundles.
- Store OAuth tokens server-side when you request scoped Agentuity access.

<Callout type="info" title="v2 frontend auth remains available for existing apps">
The v2 `@agentuity/auth/react` provider and hooks are not the default v3 frontend path. Existing v2 apps can keep using the v2 docs while new v3 apps use framework-owned auth or Agentuity OIDC routes.
</Callout>
